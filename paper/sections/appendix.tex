\section{Proof Details and Verification Notes}

\subsection{Global price continuation}
Fix an upstream history and qualities $(A_i,A_j)$. In the region where both products have positive demand, firm $i$'s profit is
\begin{equation}
 \pi_i(p_i;p_j)=\frac{p_i\left[A_i-p_i-\rho(A_j-p_j)\right]}{1-\rho^2},
 \label{eq:activeprofitappendix}
\end{equation}
which is strictly concave in $p_i$. The two active-region first-order conditions yield the candidate in \eqref{eq:price}.

Condition~\eqref{eq:R} implies, for every feasible pair of qualities,
\begin{equation}
 \frac{A_i}{A_j}
 <\frac{2}{\rho(3-\rho^2)}
 \quad\text{and}\quad
 \frac{A_i}{A_j}
 >\frac{\rho(3-\rho^2)}{2}>\rho.
 \label{eq:qualityratioappendix}
\end{equation}
The second inequality shows that a product cannot be inactive in a price equilibrium. If product $i$ had zero demand while $p_j\geq0$, then at a sufficiently small positive price $p_i$ its active-demand numerator would satisfy
\begin{equation}
 A_i-p_i-\rho(A_j-p_j)
 > A_i-\rho A_j-p_i>0
\end{equation}
for small enough $p_i>0$. Firm $i$ would then earn strictly positive profit, contradicting zero-demand equilibrium behavior. Hence no separate one-active-product or zero-demand Nash equilibrium exists.

It remains to rule out a profitable unilateral deviation from the active candidate into a region that forecloses the rival. Holding $p_j=p_j^*$, product $j$ would be driven to zero demand only if firm $i$ set
\begin{equation}
 p_i\leq A_i-\frac{A_j-p_j^*}{\rho}.
 \label{eq:foreclosurethreshold}
\end{equation}
Substituting \eqref{eq:price}, the right-hand side of \eqref{eq:foreclosurethreshold} is negative exactly when
\begin{equation}
 \frac{A_i}{A_j}<\frac{2}{\rho(3-\rho^2)},
\end{equation}
which follows from \eqref{eq:R}. Thus no feasible nonnegative deviation can foreclose the rival. A deviation high enough to make the deviator's own demand zero yields zero profit and is dominated by the positive active-equilibrium profit. Every positive-profit deviation therefore remains in the active region, where \eqref{eq:activeprofitappendix} is strictly concave and uniquely maximized by \eqref{eq:price}. The active candidate is consequently the unique global pure-strategy Nash equilibrium of every price subgame.

\subsection{General-technology allocation monotonicity and corners}
Let $g$ be differentiable, increasing, and strictly concave on $[0,E]$. Define
\begin{align}
 f_F(x,b)&=(1-\nu b)g(x)+g(E-x),\\
 f_S(x,b)&=(1+b)g(x)+g(E-x).
\end{align}
Both objectives are strictly concave in $x$, so each has a unique maximizer on $[0,E]$.

Take $b_2>b_1$. For the private problem,
\begin{equation}
 f_F(x,b_2)-f_F(x,b_1)=-\nu(b_2-b_1)g(x),
 \label{eq:private-diff-order}
\end{equation}
which is nonincreasing in $x$. This is decreasing differences without any second-derivative assumption. Let $x_k=x^F(b_k)$. If $x_2>x_1$, optimality at the two policy values implies
\begin{equation}
 f_F(x_2,b_2)-f_F(x_2,b_1)
 \geq
 f_F(x_1,b_2)-f_F(x_1,b_1),
\end{equation}
whereas \eqref{eq:private-diff-order} implies the reverse inequality. Equality would make both distinct choices optimal at both policy values, contradicting uniqueness. Hence
\begin{equation}
 x^F(b_2)\leq x^F(b_1).
\end{equation}
This argument includes boundary optima.

For the coordinated symmetric problem,
\begin{equation}
 f_S(x,b_2)-f_S(x,b_1)=(b_2-b_1)g(x),
 \label{eq:coord-diff-order}
\end{equation}
which is nondecreasing in $x$. The analogous revealed-preference argument yields
\begin{equation}
 x^S(b_2)\geq x^S(b_1),
\end{equation}
again including corners.

It remains to establish strict ordering when both compared optima are interior. First note that differentiability, monotonicity, and strict concavity imply $g'(x)>0$ at every interior $x\in(0,E)$. Indeed, monotonicity implies $g'(x)\geq0$. If $g'(x_0)=0$ at an interior point, concavity makes $g'$ nonincreasing, so $g'(x)=0$ for all $x>x_0$ where the derivative is evaluated. By the mean value theorem $g$ would then be constant on a nontrivial interval, contradicting strict concavity.

Suppose the private optima at $b_1$ and $b_2$ are both interior and, contrary to strict ordering, equal the same $x$. Their first-order conditions would be
\begin{equation}
 (1-\nu b_1)g'(x)=g'(E-x),
 \qquad
 (1-\nu b_2)g'(x)=g'(E-x).
\end{equation}
Subtracting gives $\nu(b_2-b_1)g'(x)=0$, impossible because $\nu>0$, $b_2>b_1$, and $g'(x)>0$. Combining this with global nonincreasingness gives
\begin{equation}
 x^F(b_2)<x^F(b_1).
\end{equation}
The coordinated case is identical: equality of two interior optima would require
\begin{equation}
 (1+b_1)g'(x)=g'(E-x),
 \qquad
 (1+b_2)g'(x)=g'(E-x),
\end{equation}
which is impossible for $b_2>b_1$ and $g'(x)>0$. Therefore
\begin{equation}
 x^S(b_2)>x^S(b_1).
\end{equation}
No pointwise derivative sign for $x^F(b)$ or $x^S(b)$ is asserted for a general technology.

\subsection{Threshold uniqueness}
Differentiating \eqref{eq:H} with respect to $\nu$ gives
\begin{equation}
 H_\nu=3\nu^2+4\nu(y^2-4y+1)+y^2-4y+16.
\end{equation}
For $0<\nu<y<1$, because $y^2-4y+1<0$ on the frozen range and $\nu<1$,
\begin{align}
 H_\nu
 &>4(y^2-4y+1)+(y^2-4y+16)\\
 &=5(y-2)^2>0.
\end{align}
Together with $H(y,0)<0<H(y,y)$, continuity implies a unique root $\bar\nu(y)\in(0,y)$.

\subsection{Exact consumer surplus and the Bertrand welfare gap}
At a symmetric Bertrand equilibrium, $q_1=q_2=q$. Gross utility is
\begin{equation}
 2Aq-(1+\rho)q^2.
\end{equation}
Net consumer surplus subtracts expenditure $2pq$. Using \eqref{eq:symmetricpq} yields \eqref{eq:CS}; adding equilibrium profits yields \eqref{eq:W}. If a planner instead controlled quantities at a fixed symmetric quality $A$, maximizing $2Aq-(1+\rho)q^2$ gives $q^{EFF}=A/(1+\rho)$ and $W^{EFF}=A^2/(1+\rho)$. Subtracting \eqref{eq:W} gives \eqref{eq:bertrandgap}, confirming that the coordinated symmetric-R\&D benchmark with decentralized pricing is not an unconstrained first best.
